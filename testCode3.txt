#To find Nth catlan Number
.data 
    prompt: .asciiz "Please enter an integer n : "
    debug: .asciiz "Hello\n"
.text
main:
    la $a0, prompt
    li $v0, 4
    syscall
    
    li $v0, 5
    syscall
    add $a1, $v0, $zero     # Replace move with add
    
    jal CATLAN
    # $a1 for parameter 
    # $v0 for return value
    add $a0, $v0, $zero     # Replace move with add
    li $v0, 1
    syscall
    
    li $v0, 10
    syscall
    
CATLAN: 
    li $t0, 2
    slt $t1, $a1, $t0
    beq $t1, $zero, CONTINUE # Replace bne with opposite beq logic
    j EXIT                    # Jump if condition was true
    
CONTINUE:
    addi $sp, $sp, -24
    sw $ra, 20($sp)         # store return address
    sw $a1, 16($sp)         # store n
    sw $zero, 12($sp)       # store res
    sw $zero, 8($sp)        # store i
    
LOOP:
    lw $t2, 8($sp)
    lw $t3, 16($sp)
    slt $t3, $t2, $t3
    beq $t3, $zero, RETURN
    lw $t2, 8($sp)
    lw $t3, 16($sp)
    add $a1, $t2, $zero     # Replace move with add
    jal CATLAN
    sw $v0, 4($sp)
    lw $t2, 8($sp)
    lw $t3, 16($sp)
    sub $t2, $t3, $t2
    addi $t2, $t2, -1
    add $a1, $t2, $zero     # Replace move with add
    jal CATLAN
    sw $v0, 0($sp)
    
    lw $t4, 0($sp)
    lw $t5, 4($sp)
    mul $t6, $t4, $t5       # Replace mulo with mul
    
    lw $t4, 12($sp)
    add $t4, $t4, $t6
    sw $t4, 12($sp)
    
    lw $t4, 8($sp)
    addi $t4, $t4, 1
    sw $t4, 8($sp)
    j LOOP

RETURN:
    lw $t4, 12($sp)
    add $v0, $t4, $zero     # Replace move with add
    lw $ra, 20($sp)
    addi $sp, $sp, 24
    jr $ra

EXIT:
    li $v0, 1
    jr $ra
